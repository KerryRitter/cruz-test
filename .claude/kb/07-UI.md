# UI Patterns

## Stack

- **React** with React Router v7
- **Tailwind CSS** for layout and styling
- **Chakra UI** for modals, forms, toasts
- **@cruzjs/start** for shared components (StatCard, PageHeader, etc.)

## Route Files

Route files live inside features at `features/<name>/routes/`, use `export default`, and are named with the feature prefix (e.g. `notes._index.tsx`, `notes.$id.tsx`).

Routes are declared in `<feature>.routes.ts` and registered via `@Module` — **never** manually wired in `routes.ts`.

```typescript
// src/features/notes/routes/notes._index.tsx
import { trpc } from '~/trpc/client';

export default function NotesIndexPage() {
  const { data, isLoading } = trpc.notes.list.useQuery();

  if (isLoading) return <div className="p-8">Loading...</div>;

  return (
    <div className="max-w-2xl mx-auto p-8">
      <h1 className="text-2xl font-bold mb-6">My Notes</h1>
      {data?.map((note) => (
        <div key={note.id} className="p-4 border rounded mb-4">
          <h3 className="font-semibold">{note.title}</h3>
        </div>
      ))}
    </div>
  );
}
```

```typescript
// src/features/notes/notes.routes.ts
import type { CruzRouteHelpers } from '@cruzjs/core/routing';

export function notesRoutes(helpers: CruzRouteHelpers) {
  return [
    ...helpers.prefix('notes', [
      helpers.index('features/notes/routes/notes._index.tsx'),
      helpers.route(':id', 'features/notes/routes/notes.$id.tsx'),
    ]),
  ];
}
```

```typescript
// src/features/notes/notes.module.ts — reference routes here
@Module({ routes: notesRoutes })
export class NotesModule {}
```

```typescript
// src/routes.ts — add module, never wire routes manually
export default createCruzRoutes({
  dir: import.meta.dirname,
  modules: [NotesModule],
  routes: [index('routes/index.tsx')],
});
```

## tRPC Hooks

```typescript
// Query
const { data, isLoading, error, refetch } = trpc.product.list.useQuery();

// Query with input
const { data } = trpc.product.get.useQuery({ id: productId });

// Conditional
const { data } = trpc.product.get.useQuery({ id }, { enabled: !!id });

// Mutation
const createMutation = trpc.product.create.useMutation({
  onSuccess: () => { refetch(); toast({ title: 'Created', status: 'success' }); },
  onError: (err) => { toast({ title: err.message, status: 'error' }); },
});
```

## Permission-Based UI

```typescript
const canManage = currentUserRole === 'OWNER' || currentUserRole === 'ADMIN';
const canDelete = currentUserRole === 'OWNER';

return (
  <div>
    {canManage && <Button onClick={handleCreate}>Create</Button>}
    {canDelete && <Button colorScheme="red" onClick={handleDelete}>Delete</Button>}
  </div>
);
```
